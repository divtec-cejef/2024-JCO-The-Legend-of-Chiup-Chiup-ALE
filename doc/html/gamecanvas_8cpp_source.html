183bool GameCanvas::eventFilter(QObject* pWatched, QEvent* pEvent)
184{
185 // Les événements claviers doivent être traités au niveau de GameView, sinon, certaines touches
186 // auront un effet indésirable pris en charge par QGraphicsView (ou plus exactement QAbstractScrollBar, qui
187 // gère le comportement des touches fléchées et de PageUp et PageDown).
188 // Pour que l'événement ne soit plus traité plus loin (et donc soit absorbé ici), on retourne true, comme
189 // expliqué dans la doc de QObject::eventFilter() :
190 //
191 // "In your reimplementation of this function, if you want to filter the event out, i.e. stop it being handled
192 // further, return true; otherwise return false."
193 //
194 if (qobject_cast<QGraphicsView*>(pWatched)) {
195 switch (pEvent->type()) {
196 case QEvent::KeyPress: this->keyPressed(static_cast<QKeyEvent*>(pEvent)); return true;
197 case QEvent::KeyRelease: this->keyReleased(static_cast<QKeyEvent *>(pEvent)); return true;
198 default : return QObject::eventFilter(pWatched, pEvent);
199 }
200 }
201
202 // Les événements restants proviennent de GameScene.
203 // Attention : les événements de la souris, une fois traités par GameCanevas, doivent pouvoir être gérés plus
204 // loin, par exemple dans le cas où la scène intègre des QWidget. Il faut donc retourner false.
205 switch (pEvent->type()) {
206 case QEvent::GraphicsSceneMouseMove: this->mouseMoved(static_cast<QGraphicsSceneMouseEvent*>(pEvent)); return false;
207 case QEvent::GraphicsSceneMousePress: this->mouseButtonPressed(static_cast<QGraphicsSceneMouseEvent*>(pEvent)); return false;
208 case QEvent::GraphicsSceneMouseRelease: this->mouseButtonReleased(static_cast<QGraphicsSceneMouseEvent*>(pEvent)); return false;
209 default : return QObject::eventFilter(pWatched, pEvent);
210 }
211}

228void GameCanvas::keyPressed(QKeyEvent* pKeyEvent) {
229 // Supprimer ce premier test si la répétition de touche doit être signalée.
230 if (pKeyEvent->isAutoRepeat())
231 pKeyEvent->ignore();
232 else {
233 m_pGameCore->keyPressed(pKeyEvent->key());
234
235 if (pKeyEvent->modifiers()==(Qt::ShiftModifier|Qt::ControlModifier)) {
236 switch (pKeyEvent->key()) {
237 case Qt::Key_I:
238 if (m_pDetailedInfosItem)
239 m_pDetailedInfosItem->setVisible(!m_pDetailedInfosItem->isVisible());
240 break;
241 case Qt::Key_P:
242 m_tickTimer.setInterval(m_tickTimer.interval()+1);
243 qDebug() << "Tick interval set to " << m_tickTimer.interval();
244 break;
245 case Qt::Key_M:
246 m_tickTimer.setInterval(m_tickTimer.interval()-1);
247 qDebug() << "Tick interval set to " << m_tickTimer.interval();
248 break;
249 }
250 }
251 pKeyEvent->accept();
252 }
253}

303void GameCanvas::onTick() {
304 long long elapsedTime = m_lastUpdateTime.elapsed();
305
306 // On évite une division par zéro (peu probable, mais on sait jamais)
307 if (elapsedTime < 1)
308 elapsedTime = 1;
309
310 m_lastUpdateTime.start();
311
312#ifdef QT_DEBUG
313 // Statistiques
314 m_tickCount++;
315 m_totalElapsedTime += elapsedTime;
316#endif
317
318 // Tick
319 m_pGameCore->tick(elapsedTime);
320 currentScene()->tick(elapsedTime);
321
322 if (m_pDetailedInfosItem && m_pDetailedInfosItem->isVisible())
323 m_pDetailedInfosItem->setPlainText(QString("FPS : %1, Elapsed : %2ms, Tick duration : %3ms")
324 .arg(1000/elapsedTime)
325 .arg(elapsedTime)
326 .arg(m_lastUpdateTime.elapsed()));
327
328#ifdef QT_DEBUG
329 // Statistiques
330 m_statsTrigger -= elapsedTime;
331 if (m_statsTrigger < 0) {
332 //qDebug() << "Tick count :" << m_tickCount << ". Mean tick duration : " << m_totalElapsedTime / m_tickCount << ". Min duration : " << m_minTickDuration << ". Max duration : " << m_maxTickDuration;
333 resetStatistics();
334 }
335
336 if (m_lastUpdateTime.elapsed() > m_maxTickDuration)
337 m_maxTickDuration = m_lastUpdateTime.elapsed();
338 else if (m_lastUpdateTime.elapsed() < m_minTickDuration)
339 m_minTickDuration = m_lastUpdateTime.elapsed();
340#endif
341}
