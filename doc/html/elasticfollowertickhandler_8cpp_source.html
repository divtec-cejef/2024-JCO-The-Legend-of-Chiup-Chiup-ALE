40 int currentDistance = distanceToSpriteToFollow();

42 // Si la distance entre les deux sprites dépasse la distance initiale, la tension
43 // augment petit à petit, sinon elle diminue.
44 if (currentDistance > m_initialDistance)

54 // l'élastique (plus elle est grande, plus le déplacement sera important) et l'élasticité
55 // (plus elle est grande, moins le déplacement sera important)
56 float moveDistance = qAbs(currentDistance - m_initialDistance) * m_elasticTension / m_elasticity;
57

59 QLineF lineToSprite = lineToSpriteToFollow();
