399void Sprite::onNextAnimationFrame() {
400 if (m_animationList[m_currentAnimationIndex].isEmpty()) {
401 m_currentAnimationFrame = NO_CURRENT_FRAME;
402 return;
403 }
404
405 int PreviousAnimationFrame = m_currentAnimationFrame;
406 ++m_currentAnimationFrame;
407 if (m_currentAnimationFrame >= m_animationList[m_currentAnimationIndex].count()) {
408 m_currentAnimationFrame = 0;
409 if (m_emitSignalEOA)
410 emit animationFinished();
411 if (m_animateOnceOnly) {
412 m_animateOnceOnly = false;
413 stopAnimation(AnimationStopMode::IMMEDIATE_STOP);
414 }
415 if (m_animationStopLater) {
416 m_animationStopLater = false;
417 stopAnimation(AnimationStopMode::IMMEDIATE_STOP);
418 }
419 }
420 if (PreviousAnimationFrame != m_currentAnimationFrame) {
421 setPixmap(m_animationList[m_currentAnimationIndex][m_currentAnimationFrame]);
422 update();
423 }
424}
