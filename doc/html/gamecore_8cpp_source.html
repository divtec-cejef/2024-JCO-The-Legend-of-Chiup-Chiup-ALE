242void GameCore::moveBossToPlayer() {
243 if (!m_pBoss || !m_pChiup || !bossActive) return;
244
245 // Calcul de la distance entre boss et joueur
246 qreal distance = QLineF(m_pBoss->pos(), m_pChiup->pos()).length();
247
248 // Si joueur est trop loin -> on ne fait rien
249 if (distance > 500) return;
250
251 // Si joueur est dans la portée d'attaque -> boss attaque
252 if (distance < BOSS_ATTACK_RANGE) {
253 bossAttack();
254 return;
255 }
256
257 // Calcul vecteur de déplacement vers joueur
258 qreal dx = m_pChiup->pos().x() - m_pBoss->pos().x();
259 qreal dy = m_pChiup->pos().y() - m_pBoss->pos().y();
260 qreal length = sqrt(dx * dx + dy * dy);
261
262 if (length > 0) {
263 dx /= length;
264 dy /= length;
265 }
266
267 // Déplacement du boss vers joueur
268 QPointF newPosition = m_pBoss->pos() + QPointF(dx * BOSS_SPEED * 0.01, dy * BOSS_SPEED * 0.01);
269 m_pBoss->setPos(newPosition);
270}

441void GameCore::tick(long long elapsedTimeInMilliseconds) {
442 // Calcul du déplacement en fonction des touches pressées
443 qreal deltaX = 0;
444 qreal deltaY = 0;
445 if (m_keyUpPressed) deltaY -= CHIUP_SPEED;
446 if (m_keyDownPressed) deltaY += CHIUP_SPEED;
447 if (m_keyLeftPressed) deltaX -= CHIUP_SPEED;
448 if (m_keyRightPressed) deltaX += CHIUP_SPEED;
449
450 QPointF delta(deltaX * elapsedTimeInMilliseconds / 1000,
451 deltaY * elapsedTimeInMilliseconds / 1000);
452
453 // Mise à jour de la nouvelle position
454 QPointF newPosition = m_pChiup->pos() + delta;
455
456 // Vérification des limites de la scène
457 QRectF spriteBounds = m_pChiup->boundingRect().translated(newPosition);
458 if (m_pScene->isInsideScene(spriteBounds) && canMoveTo(newPosition.x(), newPosition.y())) {
459 m_pChiup->setPos(newPosition);
460 }
461
462 // Mise à jour la position de la barre de vie pour qu'elle suive le monstre
463 for (Monster* m : m_monsters) {
464 if (!m->alive) {
465 if (m->healthBar) m->healthBar->hide();
466 if (m->healthBarBackground) m->healthBarBackground->hide();
467 continue;
468 }
469
470 double ratio = static_cast<double>(m->health) / m->maxHealth;
471
472 if (m->healthBar && m->healthBarBackground) {
473 m->healthBar->setRect(0, 0, 100 * ratio, 10);
474 m->healthBar->setPos(m->sprite->pos().x(), m->sprite->pos().y() - 20);
475 m->healthBarBackground->setPos(m->sprite->pos().x(), m->sprite->pos().y() - 20);
476 m->healthBar->show();
477 m->healthBarBackground->show();
478 }
479 }
480
481 // Centre la caméra sur le personnage
482 m_pScene->centerViewOn(m_pChiup);
483
484 // Vérifier si le joueur est en collision avec le monstre
485 checkMonsterCollision();
486
487 monsterDistance();
488
489 //Dialogue
490 if (QLineF(m_pChiup->pos(), m_pNpc->pos()).length() < 100) {
491 m_pTalkHint->setVisible(true);
492 } else {
493 m_pTalkHint->setVisible(false);
494 }
495
496 moveBossToPlayer();
497
498 if (m_pPrincess) {
499 qreal distance = QLineF(m_pPrincess->pos(), m_pChiup->pos()).length();
500
501 const qreal MIN_DISTANCE = 40.0;
502 const qreal MAX_DISTANCE = 200.0;
503
504 if (distance > MIN_DISTANCE && distance < MAX_DISTANCE) {
505 qreal dx = m_pChiup->pos().x() - m_pPrincess->pos().x();
506 qreal dy = m_pChiup->pos().y() - m_pPrincess->pos().y();
507 qreal len = sqrt(dx*dx + dy*dy);
508 if (len > 0) {
509 dx /= len;
510 dy /= len;
511
512 QPointF move(dx * 200 * elapsedTimeInMilliseconds / 1000.0,
513 dy * 200 * elapsedTimeInMilliseconds / 1000.0);
514 m_pPrincess->setPos(m_pPrincess->pos() + move);
515 }
516 }
517 }

525bool GameCore::canMoveTo(qreal x, qreal y) {
526 int tileX = static_cast<int>(x + TILE_SIZE / 2) / TILE_SIZE;
527 int tileY = static_cast<int>(y + TILE_SIZE / 2) / TILE_SIZE;
528
529 qDebug() << "Checking move to (" << x << "," << y << ")";
530 qDebug() << "Corresponding tile: (" << tileX << "," << tileY << ") -> Type:" << m_map[tileY][tileX];
531
532 if (tileX < 0 || tileX >= MAP_WIDTH || tileY < 0 || tileY >= MAP_HEIGHT) {
533 return false;
534 }
535
536 int tileType = m_map[tileY][tileX];
537
538 bool canMove = (tileType == 0 || tileType == 3 || tileType == 5 || tileType == 6);
539
540 return canMove;
541}

694

697 s->addAnimationFrame(GameFramework::imagesPath() + "brickbreaker/Snake2.png");
698 s->startAnimation(200);
699 s->setPos(pos);
700 m_pScene->addSpriteToScene(s);
701
702 // Crée la barre de vie
703 QGraphicsRectItem* bg = new QGraphicsRectItem(0, 0, 100, 10);

705 bg->setZValue(1);
706 m_pScene->addItem(bg);

710 bar->setZValue(2);
711 m_pScene->addItem(bar);
712
713 // Crée le monstre
714 Monster* m = new Monster(s, 100, 1, pos);
715 m->healthBar = bar;
716 m->healthBarBackground = bg;
717 m_monsters.append(m);
718 }
